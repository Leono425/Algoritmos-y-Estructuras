Ejercicio 4:
Escribir una función/método llamado multiplicarPorPI que toma un arreglo de enteros y
devuelve un nuevo arreglo que es el resultado de multiplicar cada elemento del arreglo original
por 3.14.
*/
using System;

double multiplicarPorPi(int[] array1)
{
    double[] array2 = new double[4];
    for (int i = 0; i < array1.Length; i++) 
    {
        array2[i] = array1[i] * 3.14;
        Console.WriteLine(array2[i]);
    }
    return 0;
}

int[] arreglo1 = new int[4];

arreglo1[0] = 1;
arreglo1[1] = 6;
arreglo1[2] = 2;
arreglo1[3] = 9;


multiplicarPorPi(arreglo1);
