using System;
using System.Collections.Generic;
using static HelloWorld;

class HelloWorld
{
    static int contador_Clientes = 0;
    static int contador_Juegos = 0;

    static Cliente[] Clientes = new Cliente[6];

    static Queue<Cliente> filaClientes = new Queue<Cliente>();

    public struct Juegos
    {
        public int ID;
        public string Nombre;
        public int Stock;
        public double Precio;
    }

    public struct Cliente
    {
        public int ID;
        public int IDproductoDeseado;
        public string Nombre;
        public int Fila;
    }

    public static class Inventario
    {
        public static void CargarJuego(Juegos[] catalogo)
        {
            if (contador_Juegos >= catalogo.Length)
            {
                Console.WriteLine("No hay espacio para agregar mas juegos.");
                return;
            }

            string nombre;
            int stock;
            double precio;

            Console.WriteLine($"Ingresa el nombre del juego numero {contador_Juegos + 1}");
            nombre = Console.ReadLine();

            Console.WriteLine($"Ingresa el stock del juego numero {contador_Juegos + 1}");
            stock = int.Parse(Console.ReadLine());

            Console.WriteLine($"Ingresa el precio del juego numero {contador_Juegos + 1}");
            precio = double.Parse(Console.ReadLine());

            catalogo[contador_Juegos] = new Juegos
            {
                ID = contador_Juegos + 1,
                Nombre = nombre,
                Stock = stock,
                Precio = precio
            };

            contador_Juegos++;

            Console.WriteLine("Juego agregado correctamente.");
        }

        public static void Mostrar(Juegos[] catalogo)
        {
            Console.WriteLine("ID\t\tPRECIO\t\tSTOCK\t\tNOMBRE");

            for (int i = 0; i < contador_Juegos; i++)
            {
                Console.WriteLine(
                    $"{catalogo[i].ID}\t\t{catalogo[i].Precio}\t\t{catalogo[i].Stock}\t\t{catalogo[i].Nombre}"
                );
            }
        }

        public static void CargarCliente()
        {
            if (contador_Clientes >= Clientes.Length)
            {
                Console.WriteLine("No hay espacio para agregar mas clientes.");
                return;
            }

            string nombre;
            int idproductoDeseado;

            Console.WriteLine($"Ingresa el nombre del cliente numero {contador_Clientes + 1}");
            nombre = Console.ReadLine();

            Console.WriteLine($"Ingresa el juego comprado por el cliente numero {contador_Clientes + 1}");
            idproductoDeseado = int.Parse(Console.ReadLine());

            Clientes[contador_Clientes] = new Cliente
            {
                ID = contador_Clientes + 1,
                Nombre = nombre,
                IDproductoDeseado = idproductoDeseado,
                Fila = contador_Clientes + 1
            };

            filaClientes.Enqueue(Clientes[contador_Clientes]);

            contador_Clientes++;

            Console.WriteLine("Cliente agregado correctamente.");
            Console.WriteLine("El cliente fue agregado a la fila.");
        }

        public static void MostrarClientes()
        {
            Console.WriteLine("ID\t\tIDproductoDeseado\t\tNOMBRE\t\tFILA");

            for (int i = 0; i < contador_Clientes; i++)
            {
                Console.WriteLine(
                    $"{Clientes[i].ID}\t\t{Clientes[i].IDproductoDeseado}\t\t{Clientes[i].Nombre}\t\t{Clientes[i].Fila}"
                );
            }
        }

        public static void AtenderCliente()
        {
            if (filaClientes.Count == 0)
            {
                Console.WriteLine("No hay clientes para atender.");
                return;
            }

            Cliente clienteAtendido = filaClientes.Dequeue();

            Console.WriteLine("--- CLIENTE ATENDIDO ---");
            Console.WriteLine($"ID: {clienteAtendido.ID}");
            Console.WriteLine($"Nombre: {clienteAtendido.Nombre}");
            Console.WriteLine($"Juego deseado: {clienteAtendido.IDproductoDeseado}");

            Queue<Cliente> nuevaFila = new Queue<Cliente>();

            int posicion = 1;

            foreach (Cliente cliente in filaClientes)
            {
                Cliente clienteActualizado = cliente;

                clienteActualizado.Fila = posicion;

                nuevaFila.Enqueue(clienteActualizado);

                posicion++;
            }

            filaClientes = nuevaFila;

            Console.WriteLine("Cliente atendido correctamente.");
        }

        public static void MostrarFila()
        {
            if (filaClientes.Count == 0)
            {
                Console.WriteLine("La fila esta vacia.");
                return;
            }

            Console.WriteLine("--- FILA DE CLIENTES ---");

            int posicion = 1;

            foreach (Cliente cliente in filaClientes)
            {
                Console.WriteLine(
                    $"Posicion: {posicion} | ID: {cliente.ID} | Nombre: {cliente.Nombre} | Juego: {cliente.IDproductoDeseado}"
                );

                posicion++;
            }
        }
    }

    static void Main()
    {
        Console.WriteLine("¿Cuantas estanterias tiene?");
        int Estanterias = int.Parse(Console.ReadLine());

        Console.WriteLine("¿Cuantos juegos pueden estar en una estanteria?");
        int Espacio_E = int.Parse(Console.ReadLine());

        Juegos[] catalogo = new Juegos[Estanterias * Espacio_E];

        opciones(catalogo);
    }

    public static void opciones(Juegos[] catalogo)
    {
        int opcion;

        do
        {
            Console.WriteLine("======================");
            Console.WriteLine("1. Agregar juego");
            Console.WriteLine("2. Mostrar juegos");
            Console.WriteLine("3. Agregar cliente");
            Console.WriteLine("4. Mostrar clientes");
            Console.WriteLine("5. Atender cliente");
            Console.WriteLine("6. Mostrar fila");
            Console.WriteLine("7. Salir");
            Console.WriteLine("======================");

            Console.Write("Elige una opcion: ");
            opcion = int.Parse(Console.ReadLine());

            switch (opcion)
            {
                case 1:
                    Inventario.CargarJuego(catalogo);
                    break;

                case 2:
                    Inventario.Mostrar(catalogo);
                    break;

                case 3:
                    Inventario.CargarCliente();
                    break;

                case 4:
                    Inventario.MostrarClientes();
                    break;

                case 5:
                    Inventario.AtenderCliente();
                    break;

                case 6:
                    Inventario.MostrarFila();
                    break;

                case 7:
                    Console.WriteLine("Saliendo del programa...");
                    break;

                default:
                    Console.WriteLine("Opcion no valida.");
                    break;
            }

        } while (opcion != 7);
    }
}
