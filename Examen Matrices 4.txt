int[,] matris = {{ 1, 2, 3},
                 { 4, 5, 6},
                 { 7, 8, 9}};

int FilaM(int[,] matriz) 
{
    int aux = 0;
    for (int i = 0; i < matriz.GetLength(0); i++)
    {
        for (int j = 0; j < matriz.GetLength(1); j++)
        {
            if (matriz[i, j] > aux) 
            {
                aux = matriz[i, j];
            }
        }
    }
    for (int i = 0; i < matriz.GetLength(0); i++)
    {
        for (int j = 0; j < matriz.GetLength(1); j++)
        {
            if (aux == matriz[i,j])
            {
                Console.Write($"La fila que contiene el maximo elemento de la matriz es la {i+1}");
            }
        }
    }
    return 0;
}
FilaM(matris);
