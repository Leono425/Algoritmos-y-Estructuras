int obtenerVehiculoMasEficiente(int[,] matriz, int[] array, int vehiculo)
{
    int aux = 0;
    for (int i = 0; i < matriz.GetLength(0); i++)
    {
        for (int j = 0; j < matriz.GetLength(1); j++)
        {
            if (j != 0)
            {
                if (aux < matriz[i, j])
                {
                    aux = matriz[i, j];
                }

            }
        }
    }
    for (int i = 0; i < matriz.GetLength(0); i++)
    {
        for (int j = 0; j < matriz.GetLength(1); j++)
        {
            if (j != 0)
            {
                if (aux == matriz[i, j])
                {
                    Console.Write($"El vehiculo que hizo mas entregas fue el {i} entregando {aux} pedidos el dia {j}");
                }

            }
        }
    }
    return 0;
}
