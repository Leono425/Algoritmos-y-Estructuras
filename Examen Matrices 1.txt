int[] arreglo = { 12, 4, 62, 5, 9, 2, 14, 23, 35, 48 };
int EnteroB = 1;
int BusquedaL(int[] array, int Buscado) 
{
    for (int i = 0; i < array.Length; i++) 
    {
        if (Buscado == array[i])
        {
            Console.Write($"el numero buscado se encuentra en el indice {i}");
        }
        else if (i == array.Length - 1) 
        {
            Console.Write("el numero buscado no se encuentra en el arreglo");
        }
    }
    return 0;
}
BusquedaL(arreglo, EnteroB);
