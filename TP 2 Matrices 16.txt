int[,] matris1 = {
    { 6, 2},
    { 1, 7},
    };
int[,] matris2 = {
    { 3, -5},
    { 2, 6},
    };
int resta_matriz(int[,] matriz1 , int[,] matriz2)
{
    Console.WriteLine(matriz1.GetLength(0));
    int aux = 0;
    int[,] matriz3 = new int[2, 2]; 
    for (int i = 0; i < matriz1.GetLength(0); i++)
    {
        for (int j = 0; j < matriz1.GetLength(1); j++)
        {
            matriz3[i,j] = matriz1[i, j] - matriz2[i,j];       
        }
    }

    for (int i = 0; i < matriz3.GetLength(0); i++)
    {
        Console.WriteLine();
        for (int j = 0; j < matriz3.GetLength(1); j++)
        {
            Console.Write($"{matriz3[i, j]} ");
        }
    }
    return 0;
}
resta_matriz(matris1, matris2);
