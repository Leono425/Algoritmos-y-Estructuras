Escribir una función/método llamado ordenar que toma un arreglo de enteros. El método
debe devolver el mismo arreglo que tomó como parámetro, pero con sus valores ordenados
de manera ascendente.*/
using System;
double ordenar(int[] array1)
{
    for (int i = 0; i < array1.Length-1; i++)
    {
        if (array1[i] > array1[i + 1])
        {
            int aux = array1[i];
            array1[i] = array1[i + 1];
            array1[i + 1] = aux;
        }

        Console.WriteLine(array1[i]);
    }
    Console.WriteLine(array1[array1.Length-1]);

    return 0;
}

int[] arreglo1 = new int[4];

arreglo1[0] = 1;
arreglo1[1] = 6;
arreglo1[2] = 2;
arreglo1[3] = 9;


ordenar(arreglo1);
