int[] arreglo1 = { 4, 7, 2,1,5,8,6,3,10,9 };
int ordenar(int[] array1) 
{
    int aux = 0;
    for (int i = 0; i < array1.Length; i++)
    {
        for (int j = 0; j < array1.Length; j++)
        {
            if (array1[i] < array1[j])
            {
                aux = array1[j];
                array1[j] = array1[i];
                array1[i] = aux;
            }
        }
    }
    for (int k = 0; k < array1.Length; k++)
    {
        Console.Write($"{array1[k]} ");
    }
    return 0; 
}
ordenar(arreglo1);
