int[,] matris = {
    { 6, -4, 3},
    { 1, 7, 10},
    { 7, 45, 2}
    };
int rotar_Derecha(int[,] matriz)
{
    Console.WriteLine(matriz.GetLength(0));
    int aux = 0;

    for (int i = 0; i <3; i++)
    {
        for (int j = 2; j > 0; j--)
        {          
                aux = matriz[i, j];
                matriz[i, j] = matriz[i, j-1];
                matriz[i, j-1] = aux;         
        }
    }

    for (int i = 0; i < matriz.GetLength(0); i++)
    {
        Console.WriteLine();
        for (int j = 0; j < matriz.GetLength(1); j++)
        {
            Console.Write($"{matriz[i, j]} ");
        }
    }
    return 0;
}
rotar_Derecha(matris);
