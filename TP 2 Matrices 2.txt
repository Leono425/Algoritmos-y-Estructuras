int[,] matris = {
    { 1, -4},
    { 6, 7},
    };
int positivos_matriz(int[,] matriz) 
{
    int cont = 0;
    for (int i = 0; i < matriz.GetLength(0); i++)
    {
        for (int j = 0; j < matriz.GetLength(1); j++)
        {
            if (matriz[i, j] >= 0) 
            {
                cont = cont + 1;
            }
        }
    }
    return cont;
}
Console.Write(positivos_matriz(matris));
