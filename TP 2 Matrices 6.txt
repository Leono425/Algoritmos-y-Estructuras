int[,] matris = {
    { 6, -4, 3},
    { 1, 7, 10},
    { 3, 45, 2}
    };
int maximo_elementoC(int[,] matriz)
{
    int aux = 0;
    int columna = 0;
    for (int i = 0; i < matriz.GetLength(0); i++)
    {
        for (int j = 0; j < matriz.GetLength(1); j++)
        {
            if (aux < matriz[i, j])
            {
                aux = matriz[i, j];
                columna = j; 
            }
        }
    }
    return columna;
}
Console.Write(maximo_elementoC(matris));
