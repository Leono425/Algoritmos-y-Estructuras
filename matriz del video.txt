int[,] matriz = new int[5, 5];
int contador = 2;
for (int i = 0; i < matriz.GetLength(0); i++)
{
    for (int j = 0; j < matriz.GetLength(1); j++) 
    {
        if (j % 2 != 0)
        {
            matriz[i, j] = contador;
            contador = contador + 2;
        }
        else 
        {
            matriz[i, j] = 0;
        }
    }
}
