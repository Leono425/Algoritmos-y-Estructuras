int[,] matris = {
    { 6, -4, 3},
    { 1, 7, 10},
    { 7, 45, 2}
    };
int cambio_filas(int[,] matriz)
{
    int aux = 0;
    int fila1 = 2;
    int fila2 = 1;

    for (int j = 0; j < matriz.GetLength(1); j++)
    {
        aux = matriz[fila1 , j];
        matriz[fila1, j] = matriz[fila2, j];
        matriz[fila2, j] = aux;
    }
    for (int i = 0; i < matriz.GetLength(0); i++)
    {
        Console.WriteLine();
        for (int j = 0; j < matriz.GetLength(1); j++)
        {
            Console.Write($"{matriz[i, j]} ");
        }
    }
    return 0;
}
cambio_filas(matris);
