using System;

int mezclarArreglos( int[] array1, int[] array2)
{
   if ( array1.Length == array2.Length)
    {
        int[] array3 = new int[array2.Length];
        for (int i = 0; i<array2.Length; i++)
        {
           
            if (i % 2 != 0)
            {
                array3[i] = array1[i];
                Console.WriteLine(array3[i]);
            }
            else
            {
                array3[i] = array2[i];
                Console.WriteLine(array3[i]);

            }
        }
        return 0;
    }
   else
    {
        Console.WriteLine("Los arreglos no son del mismo tamaño");
        return 0;
    }

}

int[] arreglo1 = new int[4];

int[] arreglo2 = new int[4];

arreglo1[0] = 1;
arreglo1[1] = 6;

arreglo2[0] = 4;
