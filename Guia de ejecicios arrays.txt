Nota: no debería ser necesario duplicar el código para calcular el producto escalar ni el

using System;
int productoEscalarConInverso(int[] array1, int[] array2)
{
    int[] array3 = new int[array1.Length];
    for (int i = 0; i < array1.Length; i++) 
    {
        array3[i] = array1[i] * array2[array2.Length - 1 - i];
        Console.WriteLine(array3[i]);
    }
    return 0;
}

int[] arreglo1 = new int[4];
int[] arreglo2 = new int[4];

arreglo2[0] = 6;
arreglo2[1] = 3;
arreglo2[2] = 7;
arreglo2[3] = 5;

arreglo1[0] = 1;
arreglo1[1] = 6;
arreglo1[2] = 2;
arreglo1[3] = 9;

productoEscalarConInverso(arreglo1, arreglo2);
