int[,] matris = {{ 1, 2, 3},
                 { 4, 5, 6},
                 { 7, 8, 9}};

int EnteroB = 9;
int Busqueda(int[,] matriz, int Buscado) 
{
    bool encontrado = false;
    for (int i = 0; i < matriz.GetLength(0); i++)
    {
        for (int j = 0; j < matriz.GetLength(1); j++)
        {
            if (Buscado == matriz[i,j])
            {
                Console.Write($"el numero buscado se encuentra en la fila {i+1}, columna {j+1}");
                encontrado = true;
            }
            else if (i == matriz.GetLength(0)-1 && j == matriz.GetLength(1) - 1 && encontrado == false)
            {
                Console.Write("el numero buscado no se encuentra en el arreglo");
            }
        }
    }
    return 0;
}
Busqueda(matris, EnteroB);
