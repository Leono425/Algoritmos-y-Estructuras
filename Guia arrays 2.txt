/*
Ejercicio 2:
Escribir una función/método llamado busquedaL que toma un arreglo de enteros y un
entero. Si el entero se encuentra entre los elementos del arreglo, el método debe devolver
la posición en la que se encuentra, caso contrario debe devolver un valor que permita
identificar a quien llama a la función/método que el entero no se encuentra en el arreglo.
*/
using System;

int busc;
Console.WriteLine("Ingrese el numero a buscar");
busc = Convert.ToInt32(Console.ReadLine());

double busquedaLineal(int[] array1)
{
    for (int i = 0; i < array1.Length; i++)
    {
        if (array1[i] == busc)
        {
            Console.WriteLine($"el numero {busc} esta en la posicion {i}");
            return 0;
        }
    }
    Console.WriteLine($"el numero {busc} no se encuentra en el array");
    return 0;
}

int[] arreglo1 = new int[4];

arreglo1[0] = 1;
arreglo1[1] = 6;
arreglo1[2] = 2;
arreglo1[3] = 9;


busquedaLineal(arreglo1);
